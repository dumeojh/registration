#[참고] 향후 원본 엑셀을 직접 변환하실 때 사용할 수 있는 Python 코드
#나중에 수강신청 결과가 바뀌어서 직접 변환하셔야 할 때를 위해, 이번에 제가 사용한 파이썬 변환 코드를 남겨드립니다.


import pandas as pd

# 1. 엑셀(CSV) 불러오기
df = pd.read_csv('수강신청결과.csv', encoding='cp949')

# 2. 열(Column) 인덱스 범위 지정 (C열=2 ~ BJ열=61)
cols_1_1_elec = df.columns[2:22]   # C ~ V
cols_1_2_elec = df.columns[22:42]  # W ~ AP
cols_1_1_comm = df.columns[42:52]  # AQ ~ AZ
cols_1_2_comm = df.columns[52:62]  # BA ~ BJ

results = []

for _, row in df.iterrows():
    if pd.isna(row['번호']) or pd.isna(row['이름']):
        continue
    
    # 값이 1.0(선택)인 과목만 리스트로 추출
    s1_1_e = [col for col in cols_1_1_elec if row[col] == 1.0]
    s1_2_e = [col for col in cols_1_2_elec if row[col] == 1.0]
    s1_1_c = [col for col in cols_1_1_comm if row[col] == 1.0]
    s1_2_c = [col for col in cols_1_2_comm if row[col] == 1.0]
    
    all_subj = s1_1_e + s1_2_e + s1_1_c + s1_2_c
    
    results.append({
        '고유번호': str(int(row['번호'])),
        '이름': str(row['이름']),
        '1학년 1학기 선택': ", ".join(s1_1_e),
        '1학년 2학기 선택': ", ".join(s1_2_e),
        '1학년 1학기 공통': ", ".join(s1_1_c),
        '1학년 2학기 공통': ", ".join(s1_2_c).replace('과탐실.1', '과탐실'), # 중복컬럼 이름 자동변경 수정
        '1학년 수강과목': ", ".join(all_subj).replace('과탐실.1', '과탐실')
    })

result_df = pd.DataFrame(results)
result_df.to_csv('cleaned_수강신청결과.csv', index=False, encoding='utf-8-sig')
